cmake_minimum_required(VERSION 3.8)
project(simulation_support)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(uav_ugv_msgs REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GZ_TRANSPORT REQUIRED IMPORTED_TARGET gz-transport13)
pkg_check_modules(GZ_MSGS REQUIRED IMPORTED_TARGET gz-msgs10)
foreach(node ugv_truth sim_vehicle_poses)
  add_executable(${node} src/${node}.cpp)
  ament_target_dependencies(${node} rclcpp geometry_msgs std_msgs uav_ugv_msgs)
  target_link_libraries(${node} PkgConfig::GZ_TRANSPORT PkgConfig::GZ_MSGS)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/obstacle_scenario DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(sim_dependencies_test test/test_sim_dependencies.py)
  ament_add_pytest_test(evaluation_test test/test_evaluation_tools.py)
  ament_add_pytest_test(obstacle_scenarios_test test/test_obstacle_scenarios.py
    APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
endif()
ament_package()
